### 程序实现

  对于重力补偿来说，我们最终会得到工具质量和质心，传感器零点和机器人安装倾角。但是实际上，考虑两种情况，第一个是机械臂固定位置，这种情况下，我们在机械臂正装的时候都不用考虑机械臂安装倾斜角的误差，因为机械安装上的误差一般来讲要低于六维力传感器的误差，所以通过六维力传感器去矫正机械安装误差是没有必要的。另外一种情况是机械臂连接在机器人上，那么我们只需要去考虑$$_{}^{\text{b}}\text{T}_s$$,因为从$$_{}^{\text{w}}\text{T}_b$$是由机器人通过一系列的连杆动态刷新的。

```C++
#pragma once
/*
构建一个通用的重力补偿程序，重点是一个函数
输入：3个位姿，3个六维力信息。
输出：重力值，零力值，零力坐标系中心，
*/
// #include "TcPch.h"
#include "TcEigen/Dense"
// #include "CR3_controlInterfaces.h"

// #include "ButterworthFilter.h"

class ToolParameterIdentification {
public:
  

    Eigen::Vector3d force_sensor_nofilter;      //没有经过滤波的数据
    Eigen::Vector3d torque_sensor_nofilter;

    Eigen::Vector3d force_sensor;      //滤波之后的数据
    Eigen::Vector3d torque_sensor;      

    Eigen::Vector3d force_no_gravity_zero_B;      //经过重力补偿之后的数据
    Eigen::Vector3d torque_no_gravity_zero_B;

   
    void toolsRecognition(int status_num, Eigen::Matrix3d& B2S_R);

    

private:
  

    Eigen::Matrix3d real2S_R;

    Eigen::Matrix3d B2S_R_1;   //输入的三组旋转矩阵与对应的力的值
    Eigen::Vector3d S_F_1;
    Eigen::Vector3d S_T_1;
    Eigen::Matrix3d B2S_R_2;
    Eigen::Vector3d S_F_2;
    Eigen::Vector3d S_T_2;
    Eigen::Matrix3d B2S_R_3;
    Eigen::Vector3d S_F_3;
    Eigen::Vector3d S_T_3;

    Eigen::Matrix<double, 9, 6> A_T;
    Eigen::Matrix<double, 9, 1> F_T;
    Eigen::Matrix<double, 9, 6> B_T;
    Eigen::Matrix<double, 9, 1> N_T;

    Eigen::Vector3d F0;  //计算出来零力
    double G = 0;     //计算出来重力
    Eigen::Vector3d T0;      //计算出来的偏移力矩
    Eigen::Vector3d S2C_P;    //计算出来的偏移向量

   
};
#include "ForceSensor.h"

void ForceSensor::toolsRecognition(int status_num, Eigen::Matrix3d &B2S_R，)
{
    switch (status_num)
    {
    case 1:
    {
        this->B2S_R_1 = B2S_R;
        this->S_F_1 = this->force_sensor;
        this->S_T_1 = this->torque_sensor;
        break;
    }
    case 2:
    {
        this->B2S_R_2 = B2S_R;
        this->S_F_2 = this->force_sensor;
        this->S_T_2 = this->torque_sensor;
        break;
    }
    case 3:
    {
        this->B2S_R_3 = B2S_R;
        this->S_F_3 = this->force_sensor;
        this->S_T_3 = this->torque_sensor;

        this->A_T.block<3,3>(0, 0) = this->B2S_R_1.inverse();
        this->A_T.block<3,3>(3, 0) = this->B2S_R_2.inverse();
        this->A_T.block<3,3>(6, 0) = this->B2S_R_3.inverse();
        this->A_T.block<3,3>(0, 3) = Eigen::Matrix3d::Identity();
        this->A_T.block<3,3>(3, 3) = Eigen::Matrix3d::Identity();
        this->A_T.block<3,3>(6, 3) = Eigen::Matrix3d::Identity();

        this->F_T << this->S_F_1, this->S_F_2, this->S_F_3;

        Eigen::Matrix<double, 6, 6> temp_66;
        Eigen::Matrix<double, 6, 1> temp_61;
        temp_66 = this->A_T.transpose() * this->A_T;
        temp_61 = temp_66.inverse() * this->A_T.transpose() * this->F_T;
        this->F0 = temp_61.block<3,1>(3,0);

        Eigen::Vector3d temp_31;
        temp_31 = temp_61.block<3,1>(0,0);
        this->G = temp_31.norm();

        Eigen::Matrix3d temp_33;
        temp_33 << 0, this->S_F_1(2), -this->S_F_1(1),
                    -this->S_F_1(2), 0, this->S_F_1(0),
                    this->S_F_1(1), -this->S_F_1(0), 0;
        this->B_T.block<3,3>(0, 0) = temp_33;
        temp_33 << 0, this->S_F_2(2), -this->S_F_2(1),
                    -this->S_F_2(2), 0, this->S_F_2(0),
                    this->S_F_2(1), -this->S_F_2(0), 0;
        this->B_T.block<3,3>(3, 0) = temp_33;
        temp_33 << 0, this->S_F_3(2), -this->S_F_3(1),
                    -this->S_F_3(2), 0, this->S_F_3(0),
                    this->S_F_3(1), -this->S_F_3(0), 0;
        this->B_T.block<3,3>(6, 0) = temp_33;
        this->B_T.block<3,3>(0, 3) = Eigen::Matrix3d::Identity();
        this->B_T.block<3,3>(3, 3) = Eigen::Matrix3d::Identity();
        this->B_T.block<3,3>(6, 3) = Eigen::Matrix3d::Identity();

        this->N_T << this->S_T_1, this->S_T_2, this->S_T_3;

        temp_66 = this->B_T.transpose() * this->B_T;
        temp_61 = temp_66.inverse() * this->B_T.transpose() * this->N_T;
        this->S2C_P = temp_61.block<3,1>(0,0);

        this->T0(0) = temp_61(3) - this->S2C_P(2) * this->F0(1) + this->S2C_P(1) * this->F0(2);
        this->T0(1) = temp_61(4) - this->S2C_P(0) * this->F0(2) + this->S2C_P(2) * this->F0(0);
        this->T0(2) = temp_61(5) - this->S2C_P(1) * this->F0(0) + this->S2C_P(0) * this->F0(1);

        break;
    }
    default:
        break;
    }
}
```
